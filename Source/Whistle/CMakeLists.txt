# The physics engine: geometry, acoustics, the time-domain model, the tuner and the instrument files, with the
# standard library alone (no JUCE), so other apps (the iPhone app) can build it as it is.
#   add_subdirectory(<whistlesim>/Source/Whistle)   then link WhistleEngine; include "Whistle/WhistleModel.h" etc.
# WHISTLESIM_PRESET_DIR may name another folder of .whistle presets (default: this repo's Presets/).

cmake_minimum_required(VERSION 3.22)
if (CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)  # Built on its own
    project(WhistleEngine LANGUAGES CXX)
endif()

set(WHISTLE_ENGINE_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../..")
if (NOT DEFINED WHISTLESIM_PRESET_DIR)
    set(WHISTLESIM_PRESET_DIR "${WHISTLE_ENGINE_ROOT}/Presets")
endif()

# The built-in presets, embedded as byte arrays in PresetFiles.cpp, in file-name order (the list's order).
# Written at configure time; changing, adding or removing a preset file re-runs the configure on the next build.
file(GLOB WHISTLE_PRESET_FILES CONFIGURE_DEPENDS "${WHISTLESIM_PRESET_DIR}/*.whistle")
list(SORT WHISTLE_PRESET_FILES)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS ${WHISTLE_PRESET_FILES})
set(WHISTLE_PRESET_ARRAYS "")
set(WHISTLE_PRESET_TABLE "")
set(WHISTLE_PRESET_INDEX 0)
foreach(PRESET_FILE IN LISTS WHISTLE_PRESET_FILES)
    get_filename_component(PRESET_NAME "${PRESET_FILE}" NAME)
    file(READ "${PRESET_FILE}" PRESET_HEX HEX)
    string(LENGTH "${PRESET_HEX}" PRESET_HEX_LENGTH)
    math(EXPR PRESET_SIZE "${PRESET_HEX_LENGTH} / 2")
    string(REGEX REPLACE "([0-9a-f][0-9a-f])" "0x\\1," PRESET_BYTES "${PRESET_HEX}")
    string(APPEND WHISTLE_PRESET_ARRAYS "const unsigned char preset${WHISTLE_PRESET_INDEX}[] = { ${PRESET_BYTES} 0 };\n")
    string(APPEND WHISTLE_PRESET_TABLE "    { \"${PRESET_NAME}\", reinterpret_cast<const char*> (preset${WHISTLE_PRESET_INDEX}), ${PRESET_SIZE} },\n")
    math(EXPR WHISTLE_PRESET_INDEX "${WHISTLE_PRESET_INDEX} + 1")
endforeach()
set(WHISTLE_PRESET_SOURCE "${CMAKE_CURRENT_BINARY_DIR}/PresetFiles.cpp")
set(WHISTLE_PRESET_TEXT "// Generated by Source/Whistle/CMakeLists.txt from the .whistle files: do not edit.\n#include \"PresetFiles.h\"\n\nnamespace whistle\n{\nnamespace\n{\n${WHISTLE_PRESET_ARRAYS}}\n\nconst std::vector<EmbeddedFile>& presetFiles()\n{\n    static const std::vector<EmbeddedFile> files {\n${WHISTLE_PRESET_TABLE}    };\n    return files;\n}\n\n} // namespace whistle\n")
if (EXISTS "${WHISTLE_PRESET_SOURCE}")
    file(READ "${WHISTLE_PRESET_SOURCE}" WHISTLE_PRESET_OLD)
endif()
if (NOT "${WHISTLE_PRESET_OLD}" STREQUAL "${WHISTLE_PRESET_TEXT}")  # Unchanged: keep the file, so nothing rebuilds
    file(WRITE "${WHISTLE_PRESET_SOURCE}" "${WHISTLE_PRESET_TEXT}")
endif()

add_library(WhistleEngine STATIC
    Acoustics.cpp
    InstrumentFile.cpp
    JetStability.cpp
    Json.cpp
    PlayedTuning.cpp
    ScalePlan.cpp
    WhistleGeometry.cpp
    WhistleModel.cpp
    "${WHISTLE_PRESET_SOURCE}")
target_compile_features(WhistleEngine PUBLIC cxx_std_20)
# Users include "Whistle/WhistleModel.h"; the engine's own files include each other by plain name.
target_include_directories(WhistleEngine PUBLIC "${WHISTLE_ENGINE_ROOT}/Source" PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
find_package(Threads REQUIRED)  # PlayedTuning plays notes in parallel (std::async)
target_link_libraries(WhistleEngine PUBLIC Threads::Threads)
